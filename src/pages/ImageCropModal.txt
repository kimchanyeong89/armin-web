
// Helper Component for Image Cropping/Positioning
const ImageCropModal = ({ imageUrl, initialCrop, onSave, onClose }: any) => {
    const [crop, setCrop] = React.useState(initialCrop);
    const [isDragging, setIsDragging] = React.useState(false);
    const lastPos = React.useRef({ x: 0, y: 0 });

    const handleStart = (clientX: number, clientY: number) => {
        setIsDragging(true);
        lastPos.current = { x: clientX, y: clientY };
    };

    const handleMove = (clientX: number, clientY: number) => {
        if (!isDragging) return;
        const dx = clientX - lastPos.current.x;
        const dy = clientY - lastPos.current.y;
        lastPos.current = { x: clientX, y: clientY };
        setCrop((prev: any) => ({ ...prev, x: prev.x + dx, y: prev.y + dy }));
    };

    const handleEnd = () => setIsDragging(false);

    return (
        <div style={{ position: 'fixed', inset: 0, zIndex: 5000, background: 'rgba(0,0,0,0.95)', display: 'flex', flexDirection: 'column', color: 'white' }}>
            <div style={{ padding: 20, textAlign: 'center', fontWeight: 'bold' }}>이미지 위치/크기 조정</div>
            <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center', overflow: 'hidden' }}
                 onTouchStart={e => handleStart(e.touches[0].clientX, e.touches[0].clientY)}
                 onTouchMove={e => handleMove(e.touches[0].clientX, e.touches[0].clientY)}
                 onTouchEnd={handleEnd}
                 onMouseDown={e => handleStart(e.clientX, e.clientY)}
                 onMouseMove={e => handleMove(e.clientX, e.clientY)}
                 onMouseUp={handleEnd}
                 onMouseLeave={handleEnd}
            >
                <div style={{
                    width: 240, height: 240, borderRadius: '50%',
                    border: '4px solid white', overflow: 'hidden',
                    position: 'relative', background: '#333'
                }}>
                     <img
                        src={imageUrl}
                        draggable={false}
                        style={{
                            width: '100%', height: '100%', objectFit: 'cover',
                            transform: `translate(${crop.x}px, ${crop.y}px) scale(${crop.scale})`,
                            cursor: 'move', pointerEvents: 'auto',
                            userSelect: 'none'
                        }}
                     />
                </div>
            </div>
            <div style={{ padding: '0 30px 40px 30px' }}>
                <p style={{ textAlign: 'center', fontSize: 13, marginBottom: 20, color: '#aaa' }}>드래그하여 이동, 슬라이더로 확대</p>
                <div style={{ display: 'flex', alignItems: 'center', gap: 10, marginBottom: 30 }}>
                    <span style={{ fontSize: 12 }}>1x</span>
                    <input
                        type="range" min="1" max="3" step="0.05"
                        value={crop.scale}
                        onChange={e => setCrop({ ...crop, scale: parseFloat(e.target.value) })}
                        style={{ flex: 1, height: 40 }}
                    />
                    <span style={{ fontSize: 12 }}>3x</span>
                </div>
                <div style={{ display: 'flex', gap: 10 }}>
                    <button onClick={onClose} style={{ flex: 1, padding: 16, borderRadius: 16, background: '#333', color: 'white', border: 'none', fontWeight: 600, fontSize: 16 }}>취소</button>
                    <button onClick={() => onSave(crop)} style={{ flex: 1, padding: 16, borderRadius: 16, background: 'white', color: 'black', border: 'none', fontWeight: 600, fontSize: 16 }}>완료</button>
                </div>
            </div>
        </div>
    );
};
